assert #0x1B86 = 0x01     $$itoa(0) length = 1$$
assert #0x1B87 = 0x30     $$itoa(0)[0] = '0'$$

assert #0x1B89 = 0x46     $$itoa(255,16)[0] = 'F'$$
assert #0x1B8A = 0x46     $$itoa(255,16)[1] = 'F'$$
