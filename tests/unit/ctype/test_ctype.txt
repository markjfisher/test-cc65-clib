////////////////////////////////
marker $$ ctype test $$
////////////////////////////////

memory load atari "${BINARY_PATH}"
symbols load "${SYMBOLS_PATH}"
registers flush

run $init_harness until (CP = $end_init_harness OR cycle_count > 5000)
assert CP = $end_init_harness $$completed correctly$$

; isalpha (mask 0x03)
assert #0x1B83 = 0x02     $$isalpha('A') returns _U mask = 0x02$$
assert #0x1B84 = 0x01     $$isalpha('z') returns _L mask = 0x01$$
assert #0x1B85 = 0x00     $$isalpha('0') = false$$
assert #0x1B86 = 0x00     $$isalpha(' ') = false$$

; isdigit (mask 0x04)
assert #0x1B87 = 0x04     $$isdigit('5') returns _D mask = 0x04$$
assert #0x1B88 = 0x04     $$isdigit('0') returns _D mask = 0x04$$
assert #0x1B89 = 0x00     $$isdigit('A') = false$$

; isalnum (mask 0x07)
assert #0x1B8A = 0x02     $$isalnum('A') returns _U mask = 0x02$$
assert #0x1B8B = 0x04     $$isalnum('9') returns _D mask = 0x04$$
assert #0x1B8C = 0x00     $$isalnum('#') = false$$

; isspace (mask 0x60)
assert #0x1B8D = 0x20     $$isspace(' ') returns _S mask = 0x20$$
assert #0x1B8E = 0x40     $$isspace('\t') returns _WS mask = 0x40$$
assert #0x1B8F = 0x00     $$isspace('A') = false$$
assert #0x1B90 = 0x40     $$isspace('\n') returns _WS mask = 0x40$$

; islower/isupper (mask 0x01 / 0x02)
assert #0x1B91 = 0x01     $$islower('a') returns _L mask = 0x01$$
assert #0x1B92 = 0x00     $$islower('A') = false$$
assert #0x1B93 = 0x02     $$isupper('Z') returns _U mask = 0x02$$
assert #0x1B94 = 0x00     $$isupper('z') = false$$

; isxdigit (mask 0x08)
assert #0x1B95 = 0x08     $$isxdigit('f') returns _X mask = 0x08$$
assert #0x1B96 = 0x00     $$isxdigit('G') = false$$

; toupper/tolower (return character)
assert #0x1B97 = 0x41     $$toupper('a') = 'A'$$
assert #0x1B98 = 0x41     $$toupper('A') = 'A'$$
assert #0x1B99 = 0x35     $$toupper('5') = '5'$$
assert #0x1B9A = 0x7A     $$tolower('Z') = 'z'$$
assert #0x1B9B = 0x7A     $$tolower('z') = 'z'$$
assert #0x1B9C = 0x24     $$tolower('$') = '$'$$

; ispunct (inverted logic: returns 0xFF for punct, 0x00 for non-punct)
assert #0x1B9D = 0xFF     $$ispunct('.') = true (0xFF)$$
assert #0x1B9E = 0x00     $$ispunct('A') = false$$

; isprint/iscntrl
assert #0x1B9F = 0x10     $$isprint(' ') = true (0x10)$$
assert #0x1BA0 = 0x00     $$isprint(0x01) = false$$
assert #0x1BA1 = 0x10     $$isprint('~') = true (0x10)$$
assert #0x1BA2 = 0x10     $$iscntrl(0x01) = true (0x10)$$
